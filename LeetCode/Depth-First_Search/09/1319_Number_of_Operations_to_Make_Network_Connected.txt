     } 

    int makeConnected(int n, vector<vector<int>>& connections) {
        if(connections.size() < n-1) return -1;
        parent.resize(n);
        rank.resize(n);

        for(int i=0; i<n; i++){
            parent[i] = i;
        }
        int component = n;

        for(int i=0; i<connections.size(); i++){
            if(find(connections[i][0]) != find(connections[i][1])){
                Union(connections[i][0], connections[i][1]);
                component--;
            }
        }
