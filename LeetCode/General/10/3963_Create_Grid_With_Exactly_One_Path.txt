        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (i == 0) {
                    if (count != 0) {
                        grid[i][j] = '.';
                        count--;
                    } else {
                        grid[i][j] = '#';
                    }
                }

                if (i != 0 && j == 0) {
                    grid[i][j] = '#';
                }
            }
        }
