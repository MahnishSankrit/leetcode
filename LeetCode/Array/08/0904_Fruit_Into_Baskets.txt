                    ans.push_back(nums[j]);
                }else if(ans.size() == 1 && ans[0] != nums[j]){
                    ans.push_back(nums[j]);
                if(ans.empty()){
            for(int j=i; j<n; j++){
                }else if(ans.size() == 2 && ans[0] != nums[j] && ans[1] != nums[j]){
                   break;
                } 
                len = max(len , j - i + 1);
            }
        }

        return len;
