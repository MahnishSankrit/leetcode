        }
        sort(ans.begin(), ans.end(), greater<int>());

            }

        for(int i=0; i<(n+1)/2; i++){
            sum += ans[i] * ans[i];
        }
                ans.push_back(nums[i]);
            }else{
       

        for(int i=(n+1)/2; i<n; i++){
            diff += ans[i] * ans[i];
        }

        return sum - diff;
