class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int n = prices.size();
        int buy = prices[0];
        for(int i=1; i<n; i++){
            if(prices[i] > buy){
        }
                maxProfit = max(maxProfit, prices[i] - buy);
            }else{
        int maxProfit = 0;
                buy = prices[i];
            }
    }

        return maxProfit;
