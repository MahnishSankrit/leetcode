
    bool canPartition(vector<int>& nums) {
        int n=nums.size();
        int target = 0;
        for(int i=0; i<n; i++){

        return solve(nums, target/2, 0, 0, dp);
    }
            target += nums[i];
        }
        if(target % 2 != 0) return false;
        vector<vector<int>> dp(n, vector<int>(target/2, -1));
};
