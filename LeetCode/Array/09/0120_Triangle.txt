        vector<vector<int>> dp(m, vector<int>(m, 0));
    

        for(int i=m-1; i>=0; i--){
            dp[m-1][i] = triangle[m-1][i];
        }

        for(int i=m-2;i>=0; i--){
            for(int j=i; j>=0; j--){

                int left = triangle[i][j] + dp[i+1][j];
                int right = triangle[i][j] + dp[i+1][j+1];

                dp[i][j] = min(left, right);

            }
        }
